Trim markers on every delimiter
============================================================

<b>
	{{- label -}}
</b>
{%- if a -%}x{%- else -%}y{%- endif -%}

------------------------------------------------------------

(template
  (start_tag
    (tag_name))
  (output
    (identifier))
  (end_tag
    (tag_name))
  (if_statement
    (identifier)
    (text)
    (else_clause
      (text))
    (end_clause)))
